#include <HX711_ADC.h>
#include <SoftwareSerial.h>
#include <LiquidCrystal_I2C.h>
#include <Servo.h>

// HX711 pinovi
const int DT_PIN = 4;
const int SCK_PIN = 5;
HX711_ADC LoadCell(DT_PIN, SCK_PIN);

// Bluetooth na pinovima 10 (RX) i 11 (TX)
SoftwareSerial Bluetooth(10, 11);

LiquidCrystal_I2C lcd(0x27, 16, 2);

// Servo objekti - koristimo imena koja si u funkcijama koristila
Servo v1; Servo v2_obrnut;
Servo s1; Servo s2;

void setup() {
  Serial.begin(57600); // Jedna brzina je dovoljna za Serial monitor[cite: 1]
  Bluetooth.begin(9600); // Standardna brzina za HC-05 module
  
  LoadCell.begin();
  LoadCell.start(2000, true); // Tare (nula) pri pokretanju[cite: 1]

  lcd.init();
  lcd.backlight();
  lcd.setCursor(0, 0);
  lcd.print("Dobar dan! -w-");
  delay(2000); 
  lcd.clear();
  lcd.print("Robot spreman!");
  
  // Povezivanje na tvoje pinove 6, 7, 8, 9
  v1.attach(6);
  v2_obrnut.attach(7);
  s1.attach(8);
  s2.attach(9);
  
  zatvoriSve(); // Postavi početni položaj
}

void loop() {
  // Slušanje naredbi iz aplikacije preko BLUETOOTHA
  if (Bluetooth.available() > 0) {
    char naredba = Bluetooth.read();
    if (naredba == 'W') { 
      otvoriSpremnik(); 
      delay(3000); 
      zatvoriSve();
    }
  }

  // Update LoadCell i detekcija težine
  if (LoadCell.update()) {
    float weight = abs(LoadCell.getData());
    if (weight > 1.0) { // Prag osjetljivosti[cite: 1]
      delay(500);
      
      // Šaljemo procjenu na mobitel preko BLUETOOTHA (ne Serial!)
      if (weight < 3.0) { 
        Bluetooth.println("1"); 
      } else { 
        Bluetooth.println("2"); 
      }

      otvoriVagu(); 
      delay(2000); 
      zatvoriSve();
    }
  }
}

// Funkcije s ispravnim imenima varijabli
void otvoriVagu() { 
  v1.write(90); 
  v2_obrnut.write(0); // Obrnuti smjer za pin 7
}
  
void otvoriSpremnik() { 
  s1.write(90); 
  s2.write(90); 
}

void zatvoriSve() { 
  v1.write(0); 
  v2_obrnut.write(90); // Obrnuti smjer za pin 7
  s1.write(0); 
  s2.write(0); 
}
